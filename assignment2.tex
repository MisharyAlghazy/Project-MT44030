\section{Shear flow per cell for design B2}
\label{sec:a2-2}



We use the multi-cell Bredt theory from the lectures. The $M_t$--$q$ relation (1st Bredt formula) and the $q$--$\theta$ relation (2nd Bredt formula, one for every cell) are
\begin{equation}
  M_t = 2\{\boldsymbol{\Omega}\}^T\{\mathbf{q}\}, \qquad
  [\mathbf{S}]\{\mathbf{q}\} = 2G\theta\{\boldsymbol{\Omega}\},
  \label{eq:B2_bredt}
\end{equation}
where all cells have the same twist rate $\theta$, because the section twists as a rigid body. Solving this system gives
\begin{equation}
  \theta = \frac{M_t}{4G}\,\frac{1}{\{\boldsymbol{\Omega}\}^T[\mathbf{S}]^{-1}\{\boldsymbol{\Omega}\}},
  \qquad
  \{\mathbf{q}\} = \frac{M_t}{2}\,\frac{[\mathbf{S}]^{-1}\{\boldsymbol{\Omega}\}}{\{\boldsymbol{\Omega}\}^T[\mathbf{S}]^{-1}\{\boldsymbol{\Omega}\}} .
  \label{eq:B2_solution}
\end{equation}
The contour matrix $[\mathbf{S}]$ has the contour integral around cell $i$ on the diagonal, and minus the integral over the wall shared by cells $i$ and $j$ off the diagonal:
\begin{equation}
  S_{ii} = \oint_i \frac{\mathrm{d}s}{t_p(s)}, \qquad
  S_{ij} = -\int_{ij} \frac{\mathrm{d}s}{t_p(s)} .
\end{equation}

\paragraph{Symmetry.}
Port cell~1 and starboard cell~1 have the same shear flow $q_1$, and they share the centre girder. In the centre girder the two flows go in opposite directions, so the net flow there is $q_1-q_1=0$. The centre girder therefore does not count in $S_{11}$, and we only have to solve the five starboard cells. For the full section every product $\{\boldsymbol{\Omega}\}^T[\mathbf{S}]^{-1}\{\boldsymbol{\Omega}\}$ is twice the value of the half section. As a check we used the same approach for B1, which gives $I_{p,B1}=108.7\,a^3t_p$ and $M_{t,B1}=75.5\,a^2t_p\tau_{max}$, the same as the values given in the assignment.

\paragraph{Cell areas}
\begin{align*}
\Omega_1 &= a\cdot a = a^2 && \text{(double bottom, } 0<y<a\text{)}\\
\Omega_2 &= 1.5a\cdot a = 1.5a^2 && \text{(double bottom, } a<y<2.5a\text{)}\\
\Omega_3 &= 1.5a\cdot a - \left(0.5a\right)^2 + \tfrac{\pi}{4}(0.5a)^2
          = \left(1.25+\tfrac{\pi}{16}\right)a^2 = 1.446\,a^2
          && \text{(double bottom with bilge)}\\
\Omega_4 &= 1.5a\cdot 2a = 3a^2 && \text{(lower wing tank)}\\
\Omega_5 &= 1.5a\cdot 2a = 3a^2 && \text{(upper wing tank)}
\end{align*}

\paragraph{Contour integrals per cell}
With the thicknesses of Table~\ref{tab:B1B2} (all terms in $a/t_p$; the bilge arc has length $\frac{\pi}{2}\cdot0.5a$):
\begin{align*}
\text{cell 1:}\quad S_{11} &= \underbrace{\tfrac{1}{3}}_{\text{bottom}}
   + \underbrace{\tfrac{1}{3}}_{\text{inner bottom}}
   + \underbrace{\tfrac{1}{3}}_{\text{side girder}} = 1,
   && S_{12} = -\tfrac{1}{3}\\
\text{cell 2:}\quad S_{22} &= \underbrace{\tfrac{1.5}{3}}_{\text{bottom}}
   + \underbrace{\tfrac{1.5}{3}}_{\text{inner bottom}}
   + \underbrace{\tfrac{1}{3}}_{\text{side girder}}
   + \underbrace{\tfrac{1}{1}}_{\text{bulkhead}} = \tfrac{7}{3},
   && S_{23} = -1\\
\text{cell 3:}\quad S_{33} &= \underbrace{\tfrac{1}{2}}_{\text{bottom}}
   + \underbrace{\tfrac{\pi/4}{2}}_{\text{bilge}}
   + \underbrace{\tfrac{0.5}{2}}_{\text{side}}
   + \underbrace{\tfrac{1.5}{3}}_{\text{inner bottom}}
   + \underbrace{\tfrac{1}{1}}_{\text{bulkhead}}
   = \tfrac{9}{4}+\tfrac{\pi}{8} = 2.643,
   && S_{34} = -\tfrac{1.5}{3}=-\tfrac{1}{2}\\
\text{cell 4:}\quad S_{44} &= \underbrace{\tfrac{1.5}{3}}_{\text{inner bottom}}
   + \underbrace{\tfrac{2}{2}}_{\text{side}}
   + \underbrace{\tfrac{1.5}{1}}_{\text{stringer deck}}
   + \underbrace{\tfrac{2}{1}}_{\text{bulkhead}} = 5,
   && S_{45} = -1.5\\
\text{cell 5:}\quad S_{55} &= \underbrace{\tfrac{1.5}{1}}_{\text{stringer deck}}
   + \underbrace{\tfrac{2}{2}}_{\text{side}}
   + \underbrace{\tfrac{1.5}{3}}_{\text{deck}}
   + \underbrace{\tfrac{2}{1}}_{\text{bulkhead}} = 5
\end{align*}

\paragraph{Contour matrix and its inverse}
\begin{equation}
[\mathbf{S}] = \frac{a}{t_p}
\begin{bmatrix}
 1          & -\tfrac13 & 0                          & 0          & 0    \\
 -\tfrac13  & \tfrac73  & -1                         & 0          & 0    \\
 0          & -1        & \tfrac94+\tfrac{\pi}{8}    & -\tfrac12  & 0    \\
 0          & 0         & -\tfrac12                  & 5          & -1.5 \\
 0          & 0         & 0                          & -1.5       & 5
\end{bmatrix}
\ \rightarrow\
[\mathbf{S}]^{-1} = \frac{t_p}{a}
\begin{bmatrix}
 1.0605 & 0.1816 & 0.0702 & 0.0077 & 0.0023 \\
 0.1816 & 0.5447 & 0.2105 & 0.0231 & 0.0069 \\
 0.0702 & 0.2105 & 0.4678 & 0.0514 & 0.0154 \\
 0.0077 & 0.0231 & 0.0514 & 0.2254 & 0.0676 \\
 0.0023 & 0.0069 & 0.0154 & 0.0676 & 0.2203
\end{bmatrix}
\label{eq:B2_S}
\end{equation}

\paragraph{Solution}
\begin{equation*}
  [\mathbf{S}]^{-1}\{\boldsymbol{\Omega}\}
  = a\,t_p\left\{1.4644,\ 1.3933,\ 1.2630,\ 0.9959,\ 0.8988\right\}^T,
\end{equation*}
\begin{equation*}
  \{\boldsymbol{\Omega}\}^T[\mathbf{S}]^{-1}\{\boldsymbol{\Omega}\}
  = 11.065\,a^3t_p \ \text{(half section)}
  \quad\rightarrow\quad
  22.131\,a^3t_p \ \text{(full section)} .
\end{equation*}
Filling this in Eq.~\eqref{eq:B2_solution} gives the twist rate
\begin{equation*}
  \theta = \frac{M_t}{4\cdot22.131\,G\,a^3t_p} = \frac{M_t}{88.52\,G\,a^3t_p}
\end{equation*}
and the shear flows
\begin{equation}
  \{\mathbf{q}\} = \frac{M_t}{2\cdot22.131\,a^3t_p}\,[\mathbf{S}]^{-1}\{\boldsymbol{\Omega}\}
  = \frac{M_t}{a^2}
  \begin{Bmatrix} 0.0331\\ 0.0315\\ 0.0285\\ 0.0225\\ 0.0203 \end{Bmatrix} .
  \label{eq:B2_q}
\end{equation}

\finalanswer{
\[
q_1 = 0.0331\,\frac{M_t}{a^2},\quad
q_2 = 0.0315\,\frac{M_t}{a^2},\quad
q_3 = 0.0285\,\frac{M_t}{a^2},\quad
q_4 = 0.0225\,\frac{M_t}{a^2},\quad
q_5 = 0.0203\,\frac{M_t}{a^2},
\]
all counter-clockwise, with $\theta = M_t/(88.5\,G\,a^3t_p)$. The mirrored port cells have the same flows.}



\section{Maximum shear stress for design B2}
\label{sec:a2-3}

The shear stress in a wall follows from the net shear flow in that wall and the plate thickness:
\begin{equation}
    \tau(s_i) = \frac{q_i}{t_p(s_i)} .
    \label{eq:shear_stress}
\end{equation}
For an outer wall (or a wall facing the open hold) $q_i$ is the flow of that cell, for a wall between two cells it is the difference between the two flows. The maximum is therefore not always in the cell with the largest flow, because the plate thickness also matters. Table~\ref{tab:B2walls} gives all walls of the starboard half.

\section{Torsional stiffness and strength}
\label{sec:a2-4}

\begin{enumerate}[label=\alph*., leftmargin=*]
  \item \textbf{Strength vs.\ stiffness.}
  Stiffness says how much a structure deforms under a load. For torsion this is $GI_p$ in $M_t = G\theta I_p$: a stiffer section twists less for the same torque. Strength says how much load the structure can take before the stress reaches the allowable value. For torsion this is the torque $M_t$ at which the highest shear stress in the section reaches $\tau_{max}$.

  \item \textbf{Torsional stiffness of B2.}
  With the full-section value of Question~\ref{sec:a2-2}:
  \begin{equation}
    I_{p,B2} = 4\{\boldsymbol{\Omega}\}^T[\mathbf{S}]^{-1}\{\boldsymbol{\Omega}\}
             = 4\cdot22.131\,a^3t_p = 88.5\,a^3t_p ,
    \qquad
    \frac{I_{p,B2}}{I_{p,B1}} = \frac{88.5}{108.7} = 0.81 .
  \end{equation}
  B2 is about 19\,\% \emph{less} stiff, even though it has the same amount of material and more cells. In B2 the longitudinal bulkhead is only $t_p$ thick instead of $2.5t_p$. The bulkhead is an outer wall of the wing tanks, which have the largest enclosed area, so a thinner bulkhead gives a much larger $\oint\mathrm{d}s/t_p$ ($4a/t_p$ instead of $1.6a/t_p$). The saved material goes into the stringer deck, the side girder and a thicker inner bottom. These are mostly walls between two cells, which only carry the small difference between two flows, so they add little to $I_p$.

  \item \textbf{Strength of B2.}
  The shear stress reaches $\tau_{max}$ first in the bulkhead of cell~4 (Question~\ref{sec:a2-3}). With Eq.~\eqref{eq:B2_solution}:
  \begin{equation}
    \tau_{max} = \frac{q_4}{t_p}
    = \frac{M_t}{2\,t_p}\cdot
      \frac{\big([\mathbf{S}]^{-1}\{\boldsymbol{\Omega}\}\big)_4}
           {\{\boldsymbol{\Omega}\}^T[\mathbf{S}]^{-1}\{\boldsymbol{\Omega}\}}
    \quad\rightarrow\quad
    M_{t,B2} = \frac{2\cdot22.131}{0.9959}\,a^2t_p\,\tau_{max}
             = 44.4\,a^2t_p\,\tau_{max} ,
  \end{equation}
  so $M_{t,B2}/M_{t,B1} = 44.4/75.5 = 0.59$. In B1 the highest stress is in the $2t_p$ side shell, while in B2 it is in the bulkhead of only $t_p$, which still carries almost the full wing tank flow. The stringer deck and side girder hardly reduce this flow.
\end{enumerate}

\finalanswer{$I_{p,B2} = 88.5\,a^3t_p$ (19\,\% lower than B1) and $M_{t,B2} = 44.4\,a^2t_p\tau_{max}$ (41\,\% lower than B1). Both are caused by the much thinner longitudinal bulkhead in B2.}

\paragraph{Twist centre.}
The pole for $\omega$ is the twist centre CT, which lies on the symmetry axis at a distance $e$ below the bottom. As in the lectures (channel section), we take moment equilibrium of the shear flow due to $F_y$ about point 3:
\begin{equation}
    z_s\,I_{zz} = \int_s \rho\,S_z(s)\,\mathrm{d}s ,
    \qquad S_z(s) = \int_A y\,\mathrm{d}A .
\end{equation}
The bottom plate goes through point 3, so there $\rho=0$. On a side shell $\rho = A/2$ and, from the free edge, $S_z(s) = \frac{A}{2}t_{sp}s$, so both side shells give $\int_s\rho\,S_z\,\mathrm{d}s = A^2t_{sp}B^2/4$. With $I_{zz} = \frac{t_{bp}A^3}{12} + 2Bt_{sp}\left(\frac{A}{2}\right)^2 = 837.3\,a^3t_p$:
\begin{equation}
    e = |z_s| = \frac{A^2B^2t_{sp}}{4\,I_{zz}} = \frac{3B^2t_{sp}}{A\,t_{bp} + 6B\,t_{sp}} = 1.672\,a .
\end{equation}

\paragraph{Sectorial coordinate.}
With $\omega(s) = -\int_s \rho(s)\,\mathrm{d}s + C$ ($\rho$ positive for counter-clockwise rotation about CT) we start in point 3 on the symmetry axis, where $\omega=0$ and $C=0$. Every part has its own local $s$, starting with the value at the end of the previous part.

\emph{Part 3--2 (bottom, port half).} Walking to $-y$ the radius vector turns counter-clockwise, so $\rho = +e$:
\begin{equation}
    \omega_{32}(s) = -e\,s , \quad 0 \le s \le \tfrac{A}{2}
    \qquad\rightarrow\qquad
    \omega_2 = -e\,\frac{A}{2} = -6.69\,a^2 .
\end{equation}

\emph{Part 2--1 (port side shell, upwards).} The radius vector turns clockwise, so $\rho = -A/2$:
\begin{equation}
    \omega_{21}(s) = \frac{A}{2}s - e\,\frac{A}{2} = \frac{A}{2}\,(s-e) , \quad 0 \le s \le B
    \qquad\rightarrow\qquad
    \omega_1 = \frac{A}{2}\,(B-e) = 13.31\,a^2 ,
\end{equation}
with $\omega = 0$ at $s = z = e$. The section is symmetric, so $\omega$ is antisymmetric: $\omega_4 = +6.69\,a^2$ and $\omega_5 = -13.31\,a^2$. This also gives $\int_A\omega\,\mathrm{d}A=0$, so no correction of $C$ is needed.

\paragraph{Sectorial area moment.}
$S_\omega(s) = -t\int_s \omega(s)\,\mathrm{d}s + C$, starting at the free edge (point 1) where $S_\omega = 0$.

\emph{Part 1--2 (port side shell, downwards).} With $\omega_{12}(s) = \frac{A}{2}(B-e-s)$:
\begin{equation}
    S_{\omega,12}(s) = t_{sp}\,\frac{A}{2}\left[\frac{s^2}{2} - (B-e)\,s\right], \quad 0 \le s \le B .
\end{equation}
The extremum is where $\omega=0$ ($z=e$): $S_{\omega,max} = -t_{sp}\frac{A}{2}\frac{(B-e)^2}{2} = -77.5\,a^3t_p$, and in point 2: $S_{\omega,2} = -58.0\,a^3t_p$.

\emph{Part 2--3 (bottom, towards the centre).} With $\omega_{23}(s) = -e\frac{A}{2} + e\,s$:
\begin{equation}
    S_{\omega,23}(s) = t_{bp}\,e\left[\frac{A}{2}\,s - \frac{s^2}{2}\right] + S_{\omega,2}, \quad 0 \le s \le \tfrac{A}{2}
    \qquad\rightarrow\qquad
    S_{\omega,3} = t_{bp}\,e\,\frac{A^2}{8} + S_{\omega,2} = +29.0\,a^3t_p .
\end{equation}
$S_\omega$ is symmetric, so the starboard half has the same values and $S_{\omega,5}=0$ at the free edge.

\section{Torsion and warping constants of design BU}
\label{sec:a2-6}

\paragraph{Torsion constant.}
BU is an open section, so we add the plates as in the lectures, $I_p = \frac13\sum_i L_i t_i^3$:
\begin{equation}
  I_p(A,B,t_{bp},t_{sp}) = \frac{1}{3}\left(A\,t_{bp}^3 + 2B\,t_{sp}^3\right)
  = \frac13\left(8\cdot6.5^3 + 2\cdot5\cdot3.5^3\right)a\,t_p^3 = 875\,a\,t_p^3 .
\end{equation}

\paragraph{Warping constant.}
$I_w = \int_A \omega^2\,\mathrm{d}A = \sum_i t_i\int_s \omega_i^2\,\mathrm{d}s$, with $\omega$ from Question~\ref{sec:a2-5}. $\omega^2$ is symmetric, so we take two times the port half:
\begin{align}
  I_{w,32} &= t_{bp}\int_0^{A/2} \left(e\,s\right)^2\mathrm{d}s = \frac{t_{bp}\,e^2A^3}{24} = 387.7\,a^5t_p ,\\
  I_{w,21} &= t_{sp}\,\frac{A^2}{4}\int_0^{B} (s-e)^2\,\mathrm{d}s
            = t_{sp}\,\frac{A^2}{4}\left(\frac{B^3}{3} - e\,B^2 + e^2B\right) = 775.3\,a^5t_p ,
\end{align}
\begin{equation}
  I_w = \frac{t_{bp}\,e^2A^3}{12} + t_{sp}\,\frac{A^2}{2}\left(\frac{B^3}{3} - e\,B^2 + e^2B\right)
  = 2\,(387.7 + 775.3)\,a^5t_p = 2.33\cdot10^3\,a^5t_p .
\end{equation}
Filling in $e = 3B^2 t_{sp}/(A t_{bp} + 6B t_{sp})$ gives the same result in $A$, $B$, $t_{bp}$ and $t_{sp}$ only:
\begin{equation}
  I_w(A,B,t_{bp},t_{sp}) = \frac{A^2 B^3 t_{sp}\left(2A t_{bp} + 3B t_{sp}\right)}{12\left(A t_{bp} + 6B t_{sp}\right)} .
\end{equation}

\finalanswer{$I_p = \tfrac{1}{3}\left(A\,t_{bp}^3 + 2B\,t_{sp}^3\right) = 875\,a\,t_p^3$, \
$I_w = \dfrac{A^2 B^3 t_{sp}\left(2A t_{bp} + 3B t_{sp}\right)}{12\left(A t_{bp} + 6B t_{sp}\right)} = 2.33\cdot10^3\,a^5 t_p$.}

\section{Effect of closing the hatch cover}
\label{sec:a2-7}

With the hatch cover ($t_{hp}=2.5t_p$) closed, the U-section becomes one closed cell with $\Omega = AB = 40a^2$. With Bredt's 2nd formula:
\begin{equation}
  I_{p,BUH} = \frac{4\,\Omega^2}{\oint \frac{\mathrm{d}s}{t_p(s)}}
  = \frac{4A^2B^2}{\dfrac{A}{t_{bp}} + \dfrac{2B}{t_{sp}} + \dfrac{A}{t_{hp}}}
  = \frac{4\cdot(40a^2)^2}{\left(\frac{8}{6.5} + \frac{10}{3.5} + \frac{8}{2.5}\right)a/t_p}
  = \frac{6400}{7.288}\,a^3t_p = 878\,a^3t_p .
\end{equation}
Compared to the open section of Question~\ref{sec:a2-6}:
\begin{equation}
  \frac{I_{p,BUH}}{I_{p,BU}} = \frac{878\,a^3t_p}{875\,a\,t_p^3} = 1.00\left(\frac{a}{t_p}\right)^2 .
\end{equation}
The open section scales with $t_p^3$ and the closed section with $a^3t_p$, which explains the factor $(a/t_p)^2$. For $a/t_p = 100$ this is about $10^4$.

\finalanswer{$I_{p,BUH} = 878\,a^3t_p$ against $I_{p,BU} = 875\,a\,t_p^3$, so closing the hatch cover increases $I_p$ by a factor $1.00\,(a/t_p)^2$.}
