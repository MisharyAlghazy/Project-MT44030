\subsection{Free warping shear stress}
\label{sec:a4-a}


\begin{enumerate}[label=\roman*., leftmargin=*]
  \item 
\begin{align}
    \tau_{xs} &= G*\theta*t_{platezone}\\
    u &= \theta\,\omega(s) \rightarrow \frac{\partial u}{\partial s} = \theta \frac{\partial\omega}{\partial s} \\
    \theta &= \frac{M_t}{GI_p}
\end{align}

Then for each member, with e as the distance to the CoR:
\begin{align}
    u_{32} = &\frac{M_t}{GI_p}*\omega_{32} = \frac{M_t}{GI_p}*e*s\\
    u_{34} = &\frac{M_t}{GI_p}*\omega_{34} = \frac{M_t}{GI_p}*-e*s\\
     u_{21} = &\frac{M_t}{GI_p}*\omega_{21} = \frac{M_t}{GI_p} (A/2)(s-e)   \\
       u_{45} = &\frac{M_t}{GI_p}*\omega_{45} = \frac{M_t}{GI_p} (A/2)(-s+e) 
\end{align}
This gives the following stress equations, where in the side plating the $\tau_{xz}$ direction is important, and in the bottom plating the $\tau_{xy}$ direction, by rewriting the open section function for shear stress according to formula 4.1.
\begin{align}
    \tau_{xz} &= G* \frac{M_t}{GI_p} *t_p  \\
    \tau_{xy} &=  G* \frac{M_t}{GI_p}* t_p
\end{align}

\subsection{Constrained warping stress}
\label{sec:a4-c}
The generalized torsion equation is as follows:
\begin{equation}
    \varphi =C_0+C_1x+C_2cosh(kx)+C_3sinh(kx)
\end{equation}
To solve this equation the following boundary conditions are applied:\\
BC1 $\rightarrow \varphi(0) = 0$ there is zero twist angle at the clamped end. \\
BC2 $\rightarrow \dot\varphi (0) = 0$ there is no warping displacement at the clamped end.\\
BC3 $\rightarrow \dot\varphi (L) = 0$ there is no warping stress at the non-clamped end.\\
BC4 $\rightarrow$ the mremote mechanical moment $M_t$ at clamped end is related to the internal one, thus $\frac{M_t}{EI_w} =C_1*K^2$ with $k = \sqrt{\frac{GI_p}{EW_p}} = 7.609*10^{-6} $\\
This gives that: $\theta = \frac{d\varphi}{dx} =  1.047*10^{-18} mm^{-1}$\\
Furthermore: \\
\begin{align}
    C_0 &= -\frac{1-cosh(\kappa L)}{sinh(\kappa L)}*\frac{M_T}{EI_wk^3} = -1.808*10^{-08} &
    C_1 &= \frac{M_T}{EI_wk^2} = 1.81*10^{-12} \\
    C_2 &= \frac{1-cosh(\kappa L)}{sinh(\kappa L)}*\frac{M_T}{EI_wk^3} = 1.808*10^{-08} &
    C_3 &= -\frac{M_T}{EI_wk^3} =  2.38*10^{-07}
\end{align}


\begin{enumerate}[label=\roman*., leftmargin=*]
  \item Analytical $\sigma_{xx}(s) =E * \frac{d\theta}{dx} *\omega(s)$ with $ \frac{d\theta}{dx} =C_2k^2cosh(kx)+C_3k^2sinh(kx) $ \\
The $\omega$ remains the same as in 4.1.2 because the section has not changed. Based on this data the following sigmas per point can be determined:

\begin{align}
   \sigma_{xx1} &= E*(A/2)(B-e)* \frac{d\theta}{dx} = 205000*4000*(3328)*1.047*10^{-18} = 2.86*10^{-06} \\
    \sigma_{xx2} &= E*z_s*(A/2)* \frac{d\theta}{dx} = 205000*1672*4000*1.047*10^{-18} =- 1.44*10^{-06}  \\
   \sigma_{xx3} &= 0 \\
    \sigma_{xx4} &= -E*z_s*(A/2)* \frac{d\theta}{dx} =1.44*10^{-06} \\
    \sigma_{xx5} &=-E*(A/2)(B-e)* \frac{d\theta}{dx} = - 2.86*10^{-06}\\
\end{align}
