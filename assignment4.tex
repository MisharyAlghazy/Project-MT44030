\subsection{Free warping shear stress}
\label{sec:a4-a}

\questionbox{Calculate and compare the analytical and numerical free warping
shear stress components $\tau_{xy}$ and $\tau_{xz}$ (through-thickness
distribution) in all structural members. Use $I_p$ obtained in
Assignment~2.6. \textbf{(5 pts)}}

\begin{enumerate}[label=\roman*., leftmargin=*]
  \item Analytical $\tau_{xy}(y,z)$ and $\tau_{xz}(y,z)$ in all structural
        members. \textbf{(1 pt)} \\ \tbd{Uitwerking}
  \item ANSYS plot of $\tau_{xy}$ and $\tau_{xz}$ in all members, with input and
        settings. \textbf{(1 pt)} \\ \tbd{screenshot en instellingen}
  \item Analytical and numerical solution in a single plot per component.
        \textbf{(1 pt)} \\ \tbd{overlay-plot}
  \item Discussion of the through-thickness distribution and maximum values.
        \textbf{(2 pts)} \\ \tbd{discussie}
\end{enumerate}
